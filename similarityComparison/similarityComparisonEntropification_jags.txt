# Similarity comparison (wrap-copy Gaussian) with Grünwald entropification
#
# Mental samples match similarityComparisonMCMC_jags.txt. The censored
# decision y ~ dinterval(dAB - dCD, 0) is replaced by a logistic in the
# signed distance difference, with one shared weight w >= 0:
#
#   P(y = 1) = 1 / (1 + exp(-w * (dAB - dCD)))
#
# y = 1 when AB is less similar than CD. A larger positive gap raises
# P(y = 1); a larger negative gap lowers it. w = 0 is chance. Trial
# latents are sampled but not monitored.

data {
  pi = 3.141592653589793
  wrapOff[1] = 0
  wrapOff[2] = -pi
  wrapOff[3] = pi
}

model {
  mu[1] ~ dunif(0, 1e-6)
  mu[2] ~ dunif(0, pi/2)
  for (j in 3:nStimuli) {
    mu[j] ~ dunif(0, pi)
  }

  sigma ~ dunif(0, pi)

  # mean 2, sd 1, truncated to [0, 20]; w multiplies the signed gap in radians
  w ~ dnorm(2, 1) T(0, 20)

  for (t in 1:nTrials) {
    nu[t] ~ dcat(c(1, 1, 1))

    xA[t] ~ dnorm(mu[a[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xB[t] ~ dnorm(mu[b[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xC[t] ~ dnorm(mu[c[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xD[t] ~ dnorm(mu[d[t]] + wrapOff[nu[t]], 1 / sigma^2)

    dAB[t] = min(abs(xA[t] - xB[t]), pi - abs(xA[t] - xB[t]))
    dCD[t] = min(abs(xC[t] - xD[t]), pi - abs(xC[t] - xD[t]))

    theta[t] = 1 / (1 + exp(-w * (dAB[t] - dCD[t])))
    y[t] ~ dbern(theta[t])
  }
}
