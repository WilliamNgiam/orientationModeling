# Similarity comparison model (Gaussian / mu wrap copies)
#
# Mental samples use dnorm centered on mu, mu-pi, or mu+pi copies
# (as in perceptual/memory reproduction), with one latent wrap choice per trial.
# Half-circle distance:
#   d = min(|x1-x2|, pi-|x1-x2|)

data {
  pi = 3.141592653589793

  wrapOff[1] = 0
  wrapOff[2] = -pi
  wrapOff[3] = pi
}

model {

  # Identifiability constraints (match original: mu[1] near zero)
  mu[1] ~ dunif(0, 1e-6)
  mu[2] ~ dunif(0, pi/2)
  for (j in 3:nStimuli) {
    mu[j] ~ dunif(0, pi)
  }

  sigma ~ dunif(0, pi)

  for (t in 1:nTrials) {
    nu[t] ~ dcat(c(1, 1, 1))

    xA[t] ~ dnorm(mu[a[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xB[t] ~ dnorm(mu[b[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xC[t] ~ dnorm(mu[c[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xD[t] ~ dnorm(mu[d[t]] + wrapOff[nu[t]], 1 / sigma^2)

    dAB[t] = min(abs(xA[t] - xB[t]), pi - abs(xA[t] - xB[t]))
    dCD[t] = min(abs(xC[t] - xD[t]), pi - abs(xC[t] - xD[t]))

    y[t] ~ dinterval(dAB[t] - dCD[t], 0)

    # posterior predictive (same wrap choice as data, cf. perceptualReproduction)
    xAp[t] ~ dnorm(mu[a[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xBp[t] ~ dnorm(mu[b[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xCp[t] ~ dnorm(mu[c[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xDp[t] ~ dnorm(mu[d[t]] + wrapOff[nu[t]], 1 / sigma^2)

    dABp[t] = min(abs(xAp[t] - xBp[t]), pi - abs(xAp[t] - xBp[t]))
    dCDp[t] = min(abs(xCp[t] - xDp[t]), pi - abs(xCp[t] - xDp[t]))
    yP[t] = step(dABp[t] - dCDp[t])

  }

}
