# Perceptual reproduction model (von Mises / doubled-angle parameterization)
#
# Line orientations live on a half-circle [0, pi). Pi ambiguity is handled by
# mapping to the full circle via phi = 2 * theta, then using dvonmises.
#
# Requires the von Mises JAGS module (../jags-vonMises).

model {
  pi2 <- 3.141592653589793 * 2
  halfPi <- pi2 / 4

  # Identifiability constraints on psychological locations (semi-circle)
  mu[1] <- 0
  mu[2] ~ dunif(0, halfPi)
  for (j in 3:nStimuli) {
    mu[j] ~ dunif(0, halfPi * 2)
  }

  # Reproduction noise: sigma is the approximate linear scale on [0, pi);
  # kappa is von Mises concentration (large kappa ~ tight distribution).
  sigma ~ dunif(0, halfPi * 2)
  kappa <- 1 / (sigma * sigma + 1e-8)

  for (r in 1:nStimuli) {
    phiMu[r] <- 2 * mu[r]
  }

  for (t in 1:nTrials) {
    yCirc[t] ~ dvonmises(phiMu[s[t]], kappa)
    yCircP[t] ~ dvonmises(phiMu[s[t]], kappa)
  }
}
